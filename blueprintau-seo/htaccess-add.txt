# Redirect /build/index.html -> /build/ (same for every folder), so Google sees one URL per page.
RewriteEngine On
RewriteCond %{THE_REQUEST} \s/+(.*/)?index\.html[\s?] [NC]
RewriteRule ^(.*/)?index\.html$ /$1 [R=301,L]
